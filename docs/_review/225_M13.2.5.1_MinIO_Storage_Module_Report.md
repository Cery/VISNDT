
### 5.2 Module Registration

```typescript
// storage.module.ts
@Global()
@Module({
  providers: [StorageService],
  exports: [StorageService],
})
export class StorageModule {}

// app.module.ts
import { StorageModule } from './storage/storage.module';
// ...
@Module({
  imports: [
    // ... 18 other modules ...
    StorageModule,  // ← Added
  ],
})
```

**Design Decision**: `@Global()` — StorageService is a cross-cutting infrastructure concern. Making it global avoids repeated imports in every module that needs file storage.

## 6. S3 Configuration

### 6.1 storage.config.ts

```typescript
export const createS3Client = (config: ConfigService): S3Client => {
  const endpoint = config.get<string>('S3_ENDPOINT');
  const region = config.get<string>('S3_REGION');
  const accessKeyId = config.get<string>('S3_ACCESS_KEY');
  const secretAccessKey = config.get<string>('S3_SECRET_KEY');

  return new S3Client({
    endpoint,
    region,
    credentials: {
      accessKeyId: accessKeyId!,
      secretAccessKey: secretAccessKey!,
    },
    forcePathStyle: true, // Required for MinIO compatibility
  });
};
```

### 6.2 Environment Variables

| Variable | Default (Dev) | Purpose |
|----------|:---:|---------|
| `S3_ENDPOINT` | `http://localhost:9000` | MinIO API endpoint |
| `S3_REGION` | `us-east-1` | AWS region (MinIO default) |
| `S3_BUCKET` | `visndt-files` | Bucket name for all uploads |
| `S3_ACCESS_KEY` | `minioadmin` | MinIO root user |
| `S3_SECRET_KEY` | `minioadmin` | MinIO root password |

**Production**: Override all values via `.env` or Docker secrets.

## 7. StorageService API

### 7.1 upload(buffer, key, mimeType)

```typescript
async upload(buffer: Buffer, key: string, mimeType: string): Promise<string>
```

| Parameter | Type | Description |
|-----------|------|-------------|
| `buffer` | `Buffer` | File content as binary buffer |
| `key` | `string` | S3/MinIO object key (path) |
| `mimeType` | `string` | Content-Type (e.g., `image/png`) |
| **Returns** | `string` | The storage key used for the upload |

**Implementation**: `s3Client.send(new PutObjectCommand({ Bucket, Key, Body, ContentType }))`

### 7.2 getSignedUrl(storageKey, expiresIn?)

```typescript
async getSignedUrl(storageKey: string, expiresIn?: number): Promise<string>
```

| Parameter | Type | Default | Description |
|-----------|------|:---:|-------------|
| `storageKey` | `string` | — | S3/MinIO object key |
| `expiresIn` | `number` | `900` (15 min) | URL expiration in seconds |
| **Returns** | `string` | — | Pre-signed download URL |

**Implementation**: `getSignedUrl(s3Client, new GetObjectCommand({ Bucket, Key }), { expiresIn })`

### 7.3 deleteObject(storageKey)

```typescript
async deleteObject(storageKey: string): Promise<void>
```

| Parameter | Type | Description |
