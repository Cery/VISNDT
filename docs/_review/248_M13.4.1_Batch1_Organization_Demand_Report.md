```typescript
// NEW METHOD
async findOnePublic(id: string) {
  const org = await this.prisma.organization.findUnique({
    where: { id },
    select: {
      id: true,
      name: true,
      type: true,
    },
  });
  if (!org) throw new NotFoundException(`Organization ${id} not found`);
  return org;
}
```

**返回字段:** 仅 `id`, `name`, `type` (不包含 members, users, invitations, status 等)
