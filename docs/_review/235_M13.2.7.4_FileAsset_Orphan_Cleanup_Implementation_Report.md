### 5.1 File: `dto/cleanup-orphans.dto.ts` (NEW)

```typescript
import { IsArray, IsUUID, ArrayMinSize, ArrayMaxSize } from 'class-validator';
import { ApiProperty } from '@nestjs/swagger';

export class CleanupOrphansDto {
  @ApiProperty({
    description: 'Array of FileAsset UUIDs to delete (max 100)',
    type: [String],
  })
  @IsArray({ message: 'ids must be an array' })
  @ArrayMinSize(1, { message: 'At least 1 ID is required' })
  @ArrayMaxSize(100, { message: 'Maximum 100 IDs allowed per request' })
  @IsUUID('4', { each: true, message: 'Each ID must be a valid UUID v4' })
  ids: string[];
}
```

### 5.2 File: `file-asset.service.ts` — New Methods

**`findOrphans()`:**
```typescript
private static readonly PLACEHOLDER_ENTITY_ID = '00000000-0000-0000-0000-000000000000';

async findOrphans() {
  return this.prisma.fileAsset.findMany({
    where: {
      entityId: FileAssetService.PLACEHOLDER_ENTITY_ID,
      media: { none: {} },
    },
    orderBy: { createdAt: 'desc' },
  });
}
```

**`cleanupOrphans(ids: string[])`:**
```typescript
async cleanupOrphans(ids: string[]): Promise<{ deleted: number; failed: string[] }> {
  let deleted = 0;
  const failed: string[] = [];

  for (const id of ids) {
    try {
      const fileAsset = await this.prisma.fileAsset.findUnique({
        where: { id },
        include: { _count: { select: { media: true } } },
      });

      if (!fileAsset) { failed.push(id); continue; }
      if (fileAsset.entityId !== FileAssetService.PLACEHOLDER_ENTITY_ID) { failed.push(id); continue; }
      if (fileAsset._count.media > 0) { failed.push(id); continue; }

      await this.delete(id); // Reuses existing DB + S3 delete
      deleted++;
    } catch (err) {
      failed.push(id);
    }
  }

  return { deleted, failed };
}
```

### 5.3 File: `file-asset.controller.ts` — New Endpoints

```typescript
// Import added: Body, CleanupOrphansDto

@Get('orphans')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles(Role.ADMIN)
@ApiBearerAuth()
@ApiOperation({ summary: 'List orphan FileAssets (ADMIN only)' })
async findOrphans() {
  const orphans = await this.service.findOrphans();
  return ApiResponse.ok(orphans, 'Orphan files retrieved');
}

@Post('orphans/cleanup')
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles(Role.ADMIN)
@ApiBearerAuth()
@ApiOperation({ summary: 'Clean up orphan FileAssets (ADMIN only, max 100 per request)' })
async cleanupOrphans(@Body() dto: CleanupOrphansDto) {
  const result = await this.service.cleanupOrphans(dto.ids);
  return ApiResponse.ok(result, 'Orphan cleanup completed');
}
```

## 6. Security Impact

### 6.1 Security Controls

| Control | Implementation |
|---|---|
| **Authentication** | `@UseGuards(JwtAuthGuard)` — JWT required |
| **Authorization** | `@Roles(Role.ADMIN)` — ADMIN-only |
| **No DELETE ALL** | No "delete all orphans" endpoint exists |
| **No auto-delete** | No cron, no scheduled task, no background job |
| **Explicit ID list** | Admin must provide explicit UUID array |
| **Max batch size** | `@ArrayMaxSize(100)` — prevents mass deletion |
| **Re-validation** | Each ID checked: exists, placeholder entityId, no media references |
| **Existing DELETE unchanged** | `DELETE /files/:id` behavior is untouched |

### 6.2 Attack Surface Analysis

| Threat | Mitigation |
|---|---|
| Unauthenticated access | `JwtAuthGuard` blocks |
| Non-admin access | `RolesGuard` + `@Roles(ADMIN)` blocks |
| Mass deletion | `ArrayMaxSize(100)` prevents large batches |
| Deleting non-orphans | Re-validation: entityId + media count |
| Deleting files still in use | `_count.media > 0` check prevents |
| Race condition (orphan created during cleanup) | Re-validation at delete time; if race loses, ID goes to failed[] |

## 7. Route Conflict Check

### 7.1 Route Order in Controller

```
GET  /files/orphans          ← LITERAL path (defined first)
POST /files/orphans/cleanup  ← LITERAL path (defined second)
GET  /files/:id/download     ← DYNAMIC path (defined after literals)
DELETE /files/:id            ← DYNAMIC path (defined after literals)
```

### 7.2 Conflict Analysis

| Request | Matches | Correct? |
|---|---|---|
| `GET /files/orphans` | `@Get('orphans')` | ✅ Literal wins |
| `POST /files/orphans/cleanup` | `@Post('orphans/cleanup')` | ✅ Literal wins |
| `GET /files/abc-123/download` | `@Get(':id/download')` | ✅ Dynamic `:id` = "abc-123" |
| `DELETE /files/abc-123` | `@Delete(':id')` | ✅ Dynamic `:id` = "abc-123" |

**No conflict.** NestJS matches literal paths before dynamic `:id` params when literals are defined first.
