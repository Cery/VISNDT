```typescript
async review(
  demandId: string,
  matchId: string,
): Promise<DemandMatch> {
  const response = (await apiClient.patch(
    `/demands/${demandId}/matches/${matchId}`,
    { status: 'REVIEWED' },
  )) as unknown as ApiResponseWrapper<DemandMatch>;
  return response.data;
}
```

- **Backend API:** Reuses existing `PATCH /demands/:id/matches/:matchId` endpoint
- **Pattern:** Consistent with existing `updateMatchStatus()` — same axios wrapper, same error handling, same return type
